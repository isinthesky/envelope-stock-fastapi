// 통계 로드
async function loadStats() {
  const hours = document.getElementById('stats_hours').value;
  const data = await getJson(`/api/v1/access-logs/stats?hours=${hours}`, null);

  if (data.success) {
    document.getElementById('stat_visitors').textContent = data.data.unique_visitors.toLocaleString();
    document.getElementById('stat_requests').textContent = data.data.total_requests.toLocaleString();

    // 경로별 통계
    const pathTbody = document.getElementById('path_stats_tbody');
    if (data.data.path_stats && data.data.path_stats.length > 0) {
      pathTbody.innerHTML = data.data.path_stats.map(s =>
        `${escapeHtml(s.path)}${s.count.toLocaleString()}`
      ).join('');
    } else {
      pathTbody.innerHTML = '데이터 없음';
    }

    // IP별 통계
    const ipTbody = document.getElementById('ip_stats_tbody');
    if (data.data.ip_stats && data.data.ip_stats.length > 0) {
      ipTbody.innerHTML = data.data.ip_stats.map(s =>
        `${escapeHtml(s.ip)}${s.count.toLocaleString()}`
      ).join('');
    } else {
      ipTbody.innerHTML = '데이터 없음';
    }
  }
}

// 로그 로드
async function loadLogs() {
  const limit = document.getElementById('log_limit').value;
  const pathFilter = document.getElementById('path_filter').value;
  const ipFilter = document.getElementById('ip_filter').value;

  let url = `/api/v1/access-logs?limit=${limit}`;
  if (pathFilter) url += `&path_filter=${encodeURIComponent(pathFilter)}`;
  if (ipFilter) url += `&ip_filter=${encodeURIComponent(ipFilter)}`;

  const data = await getJson(url, null);

  const tbody = document.getElementById('log_tbody');
  if (data.success && data.data.items.length > 0) {
    tbody.innerHTML = data.data.items.map(log => {
      const statusClass = getStatusClass(log.status_code);
      const time = log.accessed_at ? new Date(log.accessed_at).toLocaleString('ko-KR') : '-';
      const referer = log.referer ? truncate(log.referer, 30) : '-';

      return `
        ${escapeHtml(time)}
        ${escapeHtml(log.method)}
        ${escapeHtml(truncate(log.path, 40))}
        ${escapeHtml(log.ip_address)}
        ${log.status_code}
        ${log.response_time_ms}ms
        ${escapeHtml(referer)}
      `;
    }).join('');
  } else {
    tbody.innerHTML = '데이터 없음';
  }
}

async function cleanupLogs() {
  const days = document.getElementById('cleanup_days').value;
  if (!confirm(`${days}일 이전의 로그를 삭제하시겠습니까?`)) return;

  const output = document.getElementById('cleanup_output');
  output.style.display = 'block';
  await deleteJson(`/api/v1/access-logs/cleanup?days=${days}`, 'cleanup_output');
}

// 유틸리티 함수
function getStatusClass(code) {
  if (code >= 200 && code < 300) return 'status-2xx';
  if (code >= 300 && code < 400) return 'status-3xx';
  if (code >= 400 && code < 500) return 'status-4xx';
  return 'status-5xx';
}

function truncate(str, len) {
  if (!str) return '-';
  return str.length > len ? str.substring(0, len) + '...' : str;
}

function escapeHtml(str) {
  if (!str) return '';
  return str.replace(/&/g, '&').replace(//g, '>').replace(/"/g, '"');
}
