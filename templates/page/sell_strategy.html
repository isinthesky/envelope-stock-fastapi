  // 매도 전략 탭 (관심사 분리): in-page show/hide + WAI-ARIA roving tabindex
  const sellTabs = Array.from(document.querySelectorAll(".sell-tab"));
  const sellPanels = Array.from(document.querySelectorAll(".sell-panel"));

  function selectSellTab(tab, focus) {
    sellTabs.forEach((t) => {
      const on = t === tab;
      t.classList.toggle("active", on);
      t.setAttribute("aria-selected", on ? "true" : "false");
      t.setAttribute("tabindex", on ? "0" : "-1");  // roving tabindex
    });
    const target = tab.getAttribute("data-panel");
    sellPanels.forEach((p) => { p.hidden = p.id !== target; });
    if (focus) tab.focus();
  }

  sellTabs.forEach((tab, i) => {
    tab.addEventListener("click", () => selectSellTab(tab, false));
    tab.addEventListener("keydown", (e) => {
      let next = null;
      if (e.key === "ArrowRight" || e.key === "ArrowDown") next = sellTabs[(i + 1) % sellTabs.length];
      else if (e.key === "ArrowLeft" || e.key === "ArrowUp") next = sellTabs[(i - 1 + sellTabs.length) % sellTabs.length];
      else if (e.key === "Home") next = sellTabs[0];
      else if (e.key === "End") next = sellTabs[sellTabs.length - 1];
      if (next) { e.preventDefault(); selectSellTab(next, true); }  // 화살표/Home/End 이동+선택
    });
  });
