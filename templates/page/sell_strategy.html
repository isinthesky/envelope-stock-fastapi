let analysisHistory = [];

const formatNumber = (n) => {
  if (n === null || n === undefined) return '-';
  return Number(n).toLocaleString();
};

const formatDateTime = (dateStr) => {
  if (!dateStr) return '-';
  return new Date(dateStr).toLocaleString('ko-KR', { timeZone: 'Asia/Seoul' });
};

const formatTime = (dateStr) => {
  if (!dateStr) return '-';
  return new Date(dateStr).toLocaleTimeString('ko-KR', { timeZone: 'Asia/Seoul' });
};

const getRecommendationLabel = (rec) => {
  const labels = {
    'HOLD': '보유 유지',
    'WATCH': '관망',
    'WEAK_SELL': '약한 매도',
    'CONSIDER_SELL': '매도 고려',
    'SELL': '매도 권장',
    'STRONG_SELL': '강력 매도'
  };
  return labels[rec] || rec;
};

const analyzeSellSignal = async () => {
  const symbol = document.getElementById("symbol_input").value.trim();
  if (!symbol) {
    alert("종목코드를 입력해주세요.");
    return;
  }

  const stochThreshold = document.getElementById("stoch_threshold").value || 70;
  const rsiThreshold = document.getElementById("rsi_threshold").value || 70;

  document.getElementById("scan_output").style.display = "block";
  document.getElementById("scan_output").textContent = `${symbol} 분석 중... (OHLCV 데이터 조회 및 지표 계산)`;

  const url = `/api/v1/strategies/sell-signal/${symbol}?stoch_overbought=${stochThreshold}&rsi_overbought=${rsiThreshold}`;

  try {
    const response = await fetch(url);
    const data = await response.json();

    if (data.success && data.data) {
      displayResult(data.data);
      addToHistory(data.data);
      document.getElementById("scan_output").textContent = `분석 완료: ${symbol}`;
    } else {
      document.getElementById("scan_output").textContent = `오류: ${data.error || data.detail || '알 수 없는 오류'}`;
    }
  } catch (e) {
    document.getElementById("scan_output").textContent = `오류: ${e.message}`;
  }
};

  const maStatus = data.is_death_cross ? 'status-danger' : 'status-safe';

MA40 / MA160

          
${formatNumber(data.ma_short)} / ${formatNumber(data.ma_long)}

          ${data.is_death_cross ? '데드크로스' : '골든크로스'}
        

        

          
MA 갭

          
${data.ma_gap_ratio.toFixed(2)}%

        

        

          
Stochastic %K

          
${data.stoch_k.toFixed(1)}

          ${data.is_stoch_overbought ? '과매수' : '정상'}
        

        

          
Stochastic %D

          
${data.stoch_d.toFixed(1)}

        

        

          
RSI (14)

          
${data.rsi.toFixed(1)}

          ${data.is_rsi_overbought ? '과매수' : '정상'}

          ${data.sell_signal_strength} / 5
        

        

          

        

        

          보유

매도 근거

  • ${r}

const addToHistory = (data) => {
  analysisHistory.unshift(data);
  if (analysisHistory.length > 10) analysisHistory.pop();

  const tbody = document.getElementById("history_body");
  tbody.innerHTML = analysisHistory.map(item => {
    const recClass = `recommendation-${item.sell_recommendation}`;
    return `
      ${item.symbol}
      ${formatNumber(item.current_price)}
      ${getRecommendationLabel(item.sell_recommendation)}
      ${item.sell_signal_strength}/5
      ${item.stoch_k.toFixed(1)}
      ${item.rsi.toFixed(1)}
      ${formatTime(item.analyzed_at)}
    `;
  }).join('');
};

// Enter 키로 분석 실행
document.getElementById("symbol_input").addEventListener("keypress", (e) => {
  if (e.key === "Enter") analyzeSellSignal();
});
