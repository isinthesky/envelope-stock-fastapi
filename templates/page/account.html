const fmtNum = (n) => n != null ? Number(n).toLocaleString('ko-KR') : '-';
const fmtPct = (n) => n != null ? Number(n).toFixed(2) + '%' : '-';
const signClass = (n) => (n > 0 ? 'positive' : n < 0 ? 'negative' : '');

const getAccountBalance = async () => {
  const account = document.getElementById("account_no").value;
  const qs = account ? `?account_no=${account}` : "";
  const data = await getJson(`/api/v1/accounts/balance${qs}`, "account_output");
  const raw = document.getElementById("account_output");

  const b = data?.data ?? data;
  if (!b || data.success === false) {
    raw.style.display = 'block';
    document.getElementById('balance_display').style.display = 'none';
    return;
  }

  raw.style.display = 'none';
  document.getElementById('positions_display').style.display = 'none';
  const el = document.getElementById('balance_display');
  el.style.display = 'block';
  el.innerHTML = `
    

      
총 자산
${fmtNum(b.total_balance ?? b.total_eval_amount)}

      
현금
${fmtNum(b.cash_balance ?? b.deposit)}

      
주문가능
${fmtNum(b.available_amount ?? b.orderable_amount)}

      
평가손익
${fmtNum(b.total_profit_loss)}

      
손익률
${fmtPct(b.total_profit_loss_rate)}

    

  `;
};

const getPositions = async () => {
  const account = document.getElementById("account_no").value;
  const qs = account ? `?account_no=${account}` : "";
  const data = await getJson(`/api/v1/accounts/positions${qs}`, "account_output");
  const raw = document.getElementById("account_output");

  const items = data?.data?.items ?? data?.data ?? [];
  if (!Array.isArray(items) || data.success === false) {
    raw.style.display = 'block';
    document.getElementById('positions_display').style.display = 'none';
    return;
  }

  raw.style.display = 'none';
  document.getElementById('balance_display').style.display = 'none';
  const el = document.getElementById('positions_display');
  el.style.display = 'block';

  if (!items.length) {
    el.innerHTML = '
보유 포지션 없음
';
    return;
  }

  el.innerHTML = `
    
        ${items.map(p => `
          
        `).join('')}
      

      
        

          종목코드종목명수량평균단가현재가평가금액손익손익률
        

      
      

            ${p.symbol ?? '-'}
            ${p.name ?? '-'}
            ${fmtNum(p.quantity)}
            ${fmtNum(p.avg_price ?? p.purchase_price)}
            ${fmtNum(p.current_price)}
            ${fmtNum(p.eval_amount ?? p.eval_price)}
            ${fmtNum(p.profit_loss)}
            ${fmtPct(p.profit_loss_rate)}
          

    

  `;
};
