종료일

const loadHealth = async () => {
  const data = await getJson("/api/v1/ohlcv/health?freshness_days=7", "stats_output");
  if (data.success && data.data) {
    const h = data.data;
    const statusClass = h.status === "healthy" ? "health-healthy" :
                        h.status === "warning" ? "health-warning" : "health-critical";
    let html = `
${h.status?.toUpperCase() || "UNKNOWN"}
`;

    html += `

      
항목값

      
총 종목 수${h.total_symbols || 0}

      
오래된 데이터 종목${h.stale_symbols?.length || 0}개

      
최근 데이터 없는 종목${h.missing_recent?.length || 0}개

    
`;

    if (h.stale_symbols?.length > 0) {
      html += `
오래된 종목: `;
      html += h.stale_symbols.slice(0, 20).map(s => `${s}`).join("");
      if (h.stale_symbols.length > 20) html += `+${h.stale_symbols.length - 20}개`;
      html += `
`;
    }

    if (h.recommendations?.length > 0) {
      html += h.recommendations.map(r => `
${r}
`).join("");
    }

    document.getElementById("health-container").innerHTML = html;
    document.getElementById("health-container").style.display = "block";
    document.getElementById("stats-container").style.display = "none";
    document.getElementById("stats_output").style.display = "block";
  }
};

const loadSymbolGaps = () => {

// 워밍업 실행
const runWarmup = async () => {
  const symbols = document.getElementById("warmup_symbols").value.split(",").map(s => s.trim()).filter(s => s);
  if (symbols.length === 0) return alert("종목 목록을 입력하세요");

  const payload = {
    symbols: symbols,
    days: parseInt(document.getElementById("warmup_days").value) || 240,
    interval: "1d",
    force_refresh: document.getElementById("warmup_force").checked
  };
  const concurrency = document.getElementById("warmup_concurrency").value || 3;
  await postJson(`/api/v1/ohlcv/warmup?concurrency=${concurrency}`, payload, "warmup_output");
};

// 오래된 데이터 업데이트
const updateStale = async () => {
  const concurrency = document.getElementById("warmup_concurrency").value || 3;
  await postJson(`/api/v1/ohlcv/warmup/stale?freshness_days=3&concurrency=${concurrency}`, {}, "warmup_output");
};

// API 호출 예상
const estimateApiCalls = async () => {
  const symbols = document.getElementById("warmup_symbols").value.split(",").map(s => s.trim()).filter(s => s);
  if (symbols.length === 0) return alert("종목 목록을 입력하세요");

  const payload = {
    symbols: symbols,
    days: parseInt(document.getElementById("warmup_days").value) || 240,
    interval: "1d"
  };
  await postJson("/api/v1/ohlcv/warmup/estimate", payload, "warmup_output");
};

// 정리 실행
const runCleanup = async (dryRun) => {
  if (!dryRun && !confirm("정말로 오래된 데이터를 삭제하시겠습니까?")) return;

  const payload = {
    retention_days: parseInt(document.getElementById("retention_days").value) || 365,
    batch_size: parseInt(document.getElementById("batch_size").value) || 1000
  };
  await postJson(`/api/v1/ohlcv/cleanup?dry_run=${dryRun}`, payload, "cleanup_output");
};
