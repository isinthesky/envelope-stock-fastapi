const loadWsInfo = async () => {
  const data = await getJson('/ws/info', 'ws_raw_output');
  document.getElementById('ws_raw_output').style.display = 'block';

  const info = data?.data ?? data;
  if (!info) return;

  const isConnected = info.connected_clients > 0 || info.subscription_count > 0;
  const badge = document.getElementById('ws_status_badge');
  badge.style.display = 'inline-block';
  badge.className = 'ws-status-badge ' + (isConnected ? 'ws-connected' : 'ws-disconnected');
  badge.textContent = isConnected ? '연결 중' : '미연결';

  document.getElementById('ws_connections').textContent = info.connected_clients ?? '-';
  document.getElementById('ws_subscriptions').textContent = info.subscription_count ?? '-';
  document.getElementById('ws_remaining').textContent = info.remaining_slots ?? '-';
  document.getElementById('ws_kis_status').textContent = info.kis_ws_connected !== undefined ? (info.kis_ws_connected ? 'OK' : '미연결') : '-';
  document.getElementById('ws_info_grid').style.display = 'grid';

  const symbols = info.subscribed_symbols || [];
  if (symbols.length > 0) {
    document.getElementById('ws_symbols').innerHTML = symbols.map(s => `${s}`).join('');
    document.getElementById('ws_symbol_list').style.display = 'block';
  }
};
